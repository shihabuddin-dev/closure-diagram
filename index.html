Go Closures

 counter.go
func makeCounter() func() int {
    count := 0  // local var

    inner := func() int {
        count++
        return count
    }
    return inner
}

func main() {
    next := makeCounter()

    fmt.Println(next()) // ?
    fmt.Println(next()) // ?
    fmt.Println(next()) // ?
}
